# Nuka Physics v0.3 – Phase 2: PyTorch autograd.Function Skeleton + nanobind Binding

> **Master plan reference:** §3 Round 8 (Python bindings) + §3 Round 9 (autograd v0.1 skeleton)
> **Prerequisites:** v0.3 Phase 1 (perf hits gate); v0.1 Phase 7 (C ABI)
> **Blocks:** v0.3 Phase 3 (RL integration needs PyTorch tensor interop)
> **Exit criteria gate:** v0.3
> **🔒 HARD CONSTRAINT (project-wide):** GPU-only simulation. No CPU physics simulation in production code paths. See master plan §5.6.

## Goal

Build the **skeleton** of the PyTorch autograd binding. Full differentiable simulation lands in v0.5; this phase prepares the interface so v0.5 can drop in the adjoint without redesigning the Python surface.

Specifically, deliver:

1. **nanobind extension module** `nuka._core` wrapping the v0.1 C ABI.
2. **DLPack zero-copy tensor protocol** for GPU buffer interop with PyTorch.
3. **`torch.autograd.Function` skeleton** with full forward path; backward returns `None` for all gradients but the signature, tape entry hooks, and contract are in place.
4. **Caller-supplied CUDA stream**: PyTorch's `torch.cuda.current_stream()` becomes the engine's stream automatically.
5. **High-level Python `World` class** with the API shape that v0.5 / v0.7 / v1.0 will extend.

Master plan §3 Round 9 specifies "autograd skeleton at v0.1, complete at v1.0 — so tape design is constrained by autograd interface from day 1." The skeleton here lives in v0.3 actually (deferred slightly to follow v0.1 C ABI); v0.5 completes the adjoint.

## Tasks

### Task 3.2.1 — nanobind binding for Device + World

`python/src/binding.cpp`:

```cpp
#include <nanobind/nanobind.h>
#include <nanobind/stl/string_view.h>
#include "nuka/nuka.h"

namespace nb = nanobind;

NB_MODULE(_core, m) {
    nb::class_<nuka_device_t>(m, "_DeviceHandle");
    nb::class_<nuka_world_t>(m, "_WorldHandle");

    m.def("device_create", [](uint32_t gpu, uintptr_t stream_ptr) {
        nuka_device_desc_t desc{ gpu, reinterpret_cast<void*>(stream_ptr), 1 };
        nuka_device_handle h;
        auto r = nuka_device_create(&desc, &h);
        if (r != NUKA_RESULT_OK) throw std::runtime_error("device_create failed");
        return reinterpret_cast<uintptr_t>(h);
    });

    m.def("world_create_from_scene", [](uintptr_t dev, std::string_view path,
                                         uint32_t env_count, float dt) {
        nuka_world_desc_t desc{ path.data(), env_count, dt };
        nuka_world_handle h;
        auto r = nuka_world_create_from_scene(reinterpret_cast<nuka_device_handle>(dev),
                                              &desc, &h);
        if (r != NUKA_RESULT_OK) throw std::runtime_error("world_create failed");
        return reinterpret_cast<uintptr_t>(h);
    });

    m.def("world_step", [](uintptr_t w) {
        auto r = nuka_world_step(reinterpret_cast<nuka_world_handle>(w));
        if (r != NUKA_RESULT_OK) throw std::runtime_error("world_step failed");
    });

    // dlpack export (Task 3.2.2)
    m.def("world_buffer_dlpack", &world_buffer_dlpack_impl);
}
```

(`reinterpret_cast<uintptr_t>` of opaque handles is the simplest path; opaque struct types could be wrapped via nanobind too — uintptr_t keeps the binding minimal.)

### Task 3.2.2 — DLPack zero-copy buffer export

`python/src/dlpack_utils.cpp`:

```cpp
nb::object world_buffer_dlpack_impl(uintptr_t world_h, int field) {
    auto w = reinterpret_cast<nuka_world_handle>(world_h);
    nuka_buffer_view_t view;
    auto r = nuka_world_get_buffer_view(w, (nuka_state_field_t)field, &view);
    if (r != NUKA_RESULT_OK) throw std::runtime_error("buffer_view failed");

    // Construct DLManagedTensor pointing at view.device_ptr
    auto* dl = new DLManagedTensor{};
    dl->dl_tensor.data = view.device_ptr;
    dl->dl_tensor.device = DLDevice{ kDLCUDA, /*device_id*/0 };
    dl->dl_tensor.ndim = 1;
    dl->dl_tensor.dtype = dtype_from_buffer(view);
    dl->dl_tensor.shape = new int64_t[1]{ (int64_t)view.element_count };
    dl->dl_tensor.strides = nullptr;
    dl->dl_tensor.byte_offset = 0;
    dl->manager_ctx = nullptr;
    dl->deleter = [](DLManagedTensor* self) {
        delete[] self->dl_tensor.shape;
        delete self;
    };

    // Wrap as PyCapsule for torch.utils.dlpack.from_dlpack
    return nb::capsule(dl, "dltensor", [](void* p) noexcept {
        auto* self = (DLManagedTensor*)p;
        if (self->deleter) self->deleter(self);
    });
}
```

Python consumption:

```python
import torch
from torch.utils.dlpack import from_dlpack
capsule = _core.world_buffer_dlpack(world_handle, NUKA_FIELD_JOINT_POSITION)
joint_pos = from_dlpack(capsule)  # torch.Tensor, GPU, zero-copy
```

### Task 3.2.3 — High-level Python `World`

`python/nuka/world.py`:

```python
from . import _core
import torch

class World:
    def __init__(self, device: "Device", scene_path: str, num_envs: int = 1, dt: float = 1/240):
        self._device = device
        self._handle = _core.world_create_from_scene(device._handle, scene_path, num_envs, dt)
        self._num_envs = num_envs

    def step(self):
        _core.world_step(self._handle)

    def step_n(self, n: int):
        _core.world_step_n(self._handle, n)

    @property
    def joint_positions(self) -> torch.Tensor:
        capsule = _core.world_buffer_dlpack(self._handle, _core.NUKA_FIELD_JOINT_POSITION)
        return torch.utils.dlpack.from_dlpack(capsule).view(self._num_envs, -1)

    @property
    def observations(self) -> torch.Tensor:
        capsule = _core.world_buffer_dlpack(self._handle, _core.NUKA_FIELD_OBSERVATIONS)
        return torch.utils.dlpack.from_dlpack(capsule).view(self._num_envs, -1)

    # ... rewards, dones, etc — Phase 3 RL integration adds these
```

### Task 3.2.4 — Caller-supplied stream

`python/nuka/device.py`:

```python
import torch
from . import _core

class Device:
    def __init__(self, gpu_index: int = 0, cuda_stream: torch.cuda.Stream | None = None):
        if cuda_stream is None:
            cuda_stream = torch.cuda.current_stream()
        # torch.cuda.Stream wraps a cudaStream_t accessible via .cuda_stream
        stream_ptr = cuda_stream.cuda_stream
        self._handle = _core.device_create(gpu_index, stream_ptr)
        self._stream = cuda_stream

    @property
    def stream(self) -> torch.cuda.Stream:
        return self._stream
```

Now PyTorch ops and our physics share the same stream by default — no manual stream synchronization.

### Task 3.2.5 — autograd.Function skeleton

`python/nuka/autograd.py`:

```python
import torch

class _NukaPhysicsStep(torch.autograd.Function):
    """
    Skeleton: forward steps the physics. Backward returns zero grads for now;
    v0.5 fills in the real adjoint.
    """

    @staticmethod
    def forward(ctx, world_handle: int, actions: torch.Tensor) -> torch.Tensor:
        # Save context for backward (v0.5)
        ctx.world_handle = world_handle
        ctx.save_for_backward(actions)

        # Apply actions (write into world's action buffer)
        # Phase 3 plumbs the action buffer; for skeleton, accept actions as already in place

        from . import _core
        _core.world_step(world_handle)
        capsule = _core.world_buffer_dlpack(world_handle, _core.NUKA_FIELD_OBSERVATIONS)
        obs = torch.utils.dlpack.from_dlpack(capsule)
        return obs

    @staticmethod
    def backward(ctx, grad_obs: torch.Tensor):
        # v0.3: no adjoint yet; return zeros for actions
        # v0.5: call _core.world_step_backward(world_handle, grad_obs.data_ptr(), grad_actions.data_ptr())
        actions, = ctx.saved_tensors
        return None, torch.zeros_like(actions)

def step(world_handle: int, actions: torch.Tensor) -> torch.Tensor:
    """Differentiable physics step (v0.3 forward only; v0.5 backward complete)."""
    return _NukaPhysicsStep.apply(world_handle, actions)
```

The signature is locked at v0.3; v0.5 only fills in the `backward` body without changing the user-facing API.

### Task 3.2.6 — Tests

`tests/python/test_dlpack_zero_copy.py`:

```python
import torch, nuka
def test_buffer_view_is_zero_copy():
    dev = nuka.Device(0)
    w = nuka.World(dev, "examples/scenes/go2_stand.usda", num_envs=4)
    qpos1 = w.joint_positions
    initial_ptr = qpos1.data_ptr()
    w.step()
    qpos2 = w.joint_positions
    assert qpos2.data_ptr() == initial_ptr, "Tensor should view same GPU memory"
    assert qpos2.device.type == "cuda"
```

`tests/python/test_caller_stream.py`:

```python
def test_engine_uses_pytorch_stream():
    stream = torch.cuda.Stream()
    with torch.cuda.stream(stream):
        dev = nuka.Device(0, cuda_stream=stream)
        w = nuka.World(dev, "examples/scenes/go2_stand.usda", num_envs=4)
        w.step()
        # Run a torch op on the same stream; should see physics results
        # without an explicit cudaStreamSynchronize
        obs = w.observations
        result = (obs + 1.0).sum()
        # No synchronize; if streams were different we'd see stale/zero values
        assert result.item() != 0.0
```

`tests/python/test_autograd_forward_only.py`:

```python
def test_autograd_skeleton_forward_works():
    dev = nuka.Device(0)
    w = nuka.World(dev, "examples/scenes/go2_stand.usda", num_envs=4)
    actions = torch.zeros(4, w.action_dim, device="cuda", requires_grad=True)
    obs = nuka.autograd.step(w._handle, actions)
    loss = obs.sum()
    loss.backward()  # backward returns zeros; should not raise
    assert actions.grad is not None
    # v0.3 skeleton: grads are zero
    assert actions.grad.abs().sum() == 0
```
